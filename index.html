                        const binarySearch = (val, arr, lowerBound = 0, upperBound = arr.length - 1) => {
                            if (lowerBound > upperBound) {
                                return -1;
                            }
                        
                            let middlePoint = Math.floor((lowerBound + upperBound) / 2);
                            
                            if (arr[middlePoint] === val) {
                                return middlePoint;
                            }
                            else if (arr[middlePoint] > val) {
                                console.log('val is less than', arr[middlePoint]);
                                return binarySearch(val, arr, lowerBound, middlePoint - 1);
                            }
                            else if (arr[middlePoint] < val) {
                                console.log('val is greater than', arr[middlePoint]);
                                return binarySearch(val, arr, middlePoint + 1, upperBound);
                            }
                        }
                        
                    
                        public bool Binary(int[] arr, int needle, int low = 0, int hi = -1)
                        {
                            if (hi == -1)
                            {
                                hi = arr.Length - 1;
                            }
                        
                            if (low > hi)
                            {
                                return false;
                            }
                        
                            int middlePoint = (low + hi) / 2; // Integer division floors automatically
                        
                            if (arr[middlePoint] == needle)
                            {
                                // can return middlepoint if we want index
                                return true;
                            }
                            else if (needle > arr[middlePoint])
                            {
                                return Binary(arr, needle, middlePoint + 1, hi);
                            }
                            else // needle < arr[middlePoint]
                            {
                                return Binary(arr, needle, low, middlePoint - 1);
                            }
                        }

When given two crystal balls that will break if dropped from a high enough distance, determine the exact spot in which it will break in the most optimized way. This segment demonstrates breaking down a search problem without using a linear search. basically this is an array of falses, and once it crosses a threshhold all values are true (ball is breaking from this floor up). So we have to find the true value and return the index 

                        const twoCrystalBalls = (breaks) => {
                            const jumpAmount = Math.floor(Math.sqrt(breaks.length));
                        
                            let i = jumpAmount;
                            for (; i < breaks.length; i += jumpAmount) {
                                if (breaks[i]) {
                                    break;
                                }
                            }
                        
                            i -= jumpAmount;
                            for (let j = 0; j <= jumpAmount && i < breaks.length; j++, i++) {
                                if (breaks[i]) {
                                    return i;
                                }
                            }
                        
                            return -1;
                        }
