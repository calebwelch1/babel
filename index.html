                        const binarySearch = (val, arr, lowerBound = 0, upperBound = arr.length - 1) => {
                            if (lowerBound > upperBound) {
                                return -1;
                            }
                        
                            let middlePoint = Math.floor((lowerBound + upperBound) / 2);
                            
                            if (arr[middlePoint] === val) {
                                return middlePoint;
                            }
                            else if (arr[middlePoint] > val) {
                                console.log('val is less than', arr[middlePoint]);
                                return binarySearch(val, arr, lowerBound, middlePoint - 1);
                            }
                            else if (arr[middlePoint] < val) {
                                console.log('val is greater than', arr[middlePoint]);
                                return binarySearch(val, arr, middlePoint + 1, upperBound);
                            }
                        }
                        
                    
                        public static int? BinarySearch(int[] arr, int val, int lowerBound = 0, int upperBound = -1)
                        {
                            if (upperBound == -1)
                            {
                                upperBound = arr.Length - 1;
                            }
                    
                            if (lowerBound > upperBound)
                            {
                                return null;
                            }
                    
                            // Calculate middle point
                            int middlePoint = lowerBound + (upperBound - lowerBound) / 2;
                    
                            // Compare value at middle point with target
                            if (arr[middlePoint] == val)
                            {
                                return middlePoint;  // Found it, return index
                            }
                            else if (arr[middlePoint] > val)
                            {
                                // Search left half
                                return BinarySearch(arr, val, lowerBound, middlePoint - 1);
                            }
                            else
                            {
                                // Search right half
                                return BinarySearch(arr, val, middlePoint + 1, upperBound);
                            }
                        }
